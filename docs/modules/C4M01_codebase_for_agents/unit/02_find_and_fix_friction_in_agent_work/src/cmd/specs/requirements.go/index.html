  21. 21func readRequirements(reader io.Reader) ([]requirement, error) {
  22. 22 var requirements []requirement
  23. 23 seen := map[spec.RequirementID]int{}
  24. 24 scanner := bufio.NewScanner(reader)
  25. 25 line := 0
  26. 26 for scanner.Scan() {
  27. 27 line++
  28. 28 match := requirementLine.FindStringSubmatch(scanner.Text())
  29. 29 if match == nil {
  30. 30 continue
  31. 31 }
  32. 32 id, err := spec.ParseRequirementID(match[1])
  33. 33 if err != nil {
  34. 34 return nil, fmt.Errorf("line %d: %w", line, err)
  35. 35 }
  36. 36 if previous, exists := seen[id]; exists {
  37. 37 return nil, fmt.Errorf("line %d: duplicate requirement %s (first at line %d)", line, id, previous)
  38. 38 }
  39. 39 seen[id] = line
  40. 40 requirements = append(requirements, requirement{ID: id, Text: match[2]})
  41. 41 }
  42. 42 if err := scanner.Err(); err != nil {
  43. 43 return nil, err
  44. 44 }
  45. 45 if len(requirements) == 0 {
  46. 46 return nil, fmt.Errorf("no numbered requirements found")
  47. 47 }
  48. 48 return requirements, nil
  49. 49}

  51. 51func validateRequirements(requirements []requirement, features []spec.Feature) error {
  52. 52 known := make(map[spec.RequirementID]bool, len(requirements))
  53. 53 for _, requirement := range requirements {
  54. 54 known[requirement.ID] = true
  55. 55 }
  56. 56 for _, feature := range features {
  57. 57 for _, scenario := range feature.Specs {
  58. 58 if len(scenario.Requirements) == 0 {
  59. 59 return fmt.Errorf("%s / %s: missing Covers requirements", feature.Name, scenario.Name)
  60. 60 }
  61. 61 seen := map[spec.RequirementID]bool{}
  62. 62 for _, id := range scenario.Requirements {
  63. 63 if !known[id] {
  64. 64 return fmt.Errorf("%s / %s: unknown requirement %q", feature.Name, scenario.Name, id)
  65. 65 }
  66. 66 if seen[id] {
  67. 67 return fmt.Errorf("%s / %s: duplicate requirement %s", feature.Name, scenario.Name, id)
  68. 68 }
  69. 69 seen[id] = true
  70. 70 }
  71. 71 }
  72. 72 }
  73. 73 return nil
  74. 74}

  82. 82func printCoverage(out io.Writer, requirements []requirement, outcomes []scenarioOutcome) {
  83. 83 fmt.Fprintln(out, "\nRequirement interpretations (PASS does not establish completeness):")
  84. 84 writer := tabwriter.NewWriter(out, 0, 4, 2, ' ', 0)
  85. 85 fmt.Fprintln(writer, "REQUIREMENT\tSTATE\tSCENARIO\tTHEN")
  86. 86 for _, requirement := range requirements {
  87. 87 linked := false
  88. 88 for _, outcome := range outcomes {
  89. 89 if !slices.Contains(outcome.Scenario.Requirements, requirement.ID) {
  90. 90 continue
  91. 91 }
  92. 92 linked = true
  93. 93 state := "PASS"
  94. 94 if len(outcome.Failures) > 0 {
  95. 95 state = "FAIL"
  96. 96 }
  97. 97 fmt.Fprintf(writer, "%s\t%s\t%s / %s\t%s\n", requirement.ID, state, outcome.Feature, outcome.Scenario.Name, outcome.Scenario.Meaning)
  98. 98 }
  99. 99 if !linked {
  100. 100 fmt.Fprintf(writer, "%s\tNONE\t(no linked scenario)\t%s\n", requirement.ID, requirement.Text)
  101. 101 }
  102. 102 }
  103. 103 _ = writer.Flush()
  104. 104}
