  8. 8func organizerNavigationSpecs() spec.Specs {
  9. 9 var specs spec.Specs
  10. 10
  11. 11 for _, page := range []struct{ name, uri, active string }{
  12. 12 {"catalogue", "/", "/"},
  13. 13 {"creation", "/workshops/new", "/workshops/new"},
  14. 14 {"reservations", "/workshops/october/reservations", "/"},
  15. 15 } {
  16. 16 specs.Add("Organizer navigates from the "+page.name+" page", func(bld *spec.B) {
  17. 17 bld.Given(evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 1})
  18. 18 bld.When(spec.Get(page.uri)).
  19. 19 Then("the product, catalogue, and creation are reachable from the top bar").
  20. 20 Covers("W10").
  21. 21 ExpectEl("#product-link", spec.With{"href": "/", "text": "Workshop Reservations"}).
  22. 22 ExpectElementCount(`#primary-nav a[href="/"]`, 1).
  23. 23 ExpectElementCount(`#primary-nav a[href="/workshops/new"]`, 1)
  24. 24 })
  25. 25 specs.Add("Organizer locates the "+page.name+" page in navigation", func(bld *spec.B) {
  26. 26 bld.Given(evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 1})
  27. 27 bld.When(spec.Get(page.uri)).
  28. 28 Then("the current navigation section is identified").
  29. 29 Covers("W10").
  30. 30 ExpectEl("#primary-nav a.active", spec.With{"href": page.active})
  31. 31 })
  32. 32 specs.Add("Organizer follows the "+page.name+" breadcrumb", func(bld *spec.B) {
  33. 33 bld.Given(evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 1})
  34. 34 step := bld.When(spec.Get(page.uri)).
  35. 35 Covers("W10")
  36. 36 if page.uri == "/" {
  37. 37 step.Then("the catalogue has no breadcrumb back to itself").
  38. 38 ExpectNoElement(`#breadcrumbs a[href="/"]`)
  39. 39 } else {
  40. 40 step.Then("the breadcrumb returns to the workshop catalogue").
  41. 41 ExpectElementCount(`#breadcrumbs a[href="/"]`, 1)
  42. 42 }
  43. 43 })
  44. 44 }
  45. 45
  46. 46 for _, example := range []struct {
  47. 47 name string
  48. 48 events []evt.Event
  49. 49 }{
  50. 50 {"empty", nil},
  51. 51 {"populated", []evt.Event{(evt.SeatReserved{WorkshopID: "october", Email: "alice@example.com"})}},
  52. 52 } {
  53. 53 specs.Add("Organizer opens the participant link from the "+example.name+" reservations page", func(bld *spec.B) {
  54. 54 bld.Given(evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 1})
  55. 55 bld.Given(example.events...)
  56. 56 bld.When(spec.Get("/workshops/october/reservations")).
  57. 57 Then("sharing and the breadcrumb lead to the workshop without an email or a separate workshop navigation bar").
  58. 58 Covers("W2", "W10").
  59. 59 ExpectEl("#share-link", spec.With{"href": "/workshops/october"}).
  60. 60 ExpectEl(`#breadcrumbs a:not([href="/"])`, spec.With{"href": "/workshops/october"}).
  61. 61 ExpectNoElement("#workshop-nav")
  62. 62 })
  63. 63 }
  64. 64
  65. 65 return specs
