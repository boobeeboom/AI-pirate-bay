  12. 12func workshopCreationSpecs() spec.Specs {
  13. 13 var specs spec.Specs
  14. 14
  15. 15 specs.Add("Visitor opens workshop creation", func(bld *spec.B) {
  16. 16 bld.When(spec.Get("/workshops/new")).
  17. 17 Then("creation has its own form and a way back to the catalogue").
  18. 18 Covers("W1").
  19. 19 ExpectEl("form:has(#title)", spec.With{"method": "post", "action": "/workshops"}).
  20. 20 ExpectEl("form input#title", spec.With{"name": "title"}).
  21. 21 ExpectEl("form:has(#title) input#capacity", spec.With{"name": "capacity"}).
  22. 22 ExpectEl("#cancel-create", spec.With{"href": "/"})
  23. 23 })
  24. 24
  25. 25 specs.Add("Visitor corrects an invalid capacity", func(bld *spec.B) {
  26. 26 bld.When(spec.Post("/workshops", spec.Form{"title": " Event Storming ", "capacity": "0"})).
  27. 27 Then("the visitor can correct capacity without retyping the title").
  28. 28 Covers("W1", "W12").
  29. 29 ExpectStatus(http.StatusBadRequest).
  30. 30 ExpectNoEvents().
  31. 31 ExpectEl("#title", spec.With{"value": " Event Storming "}).
  32. 32 ExpectEl("#capacity", spec.With{"value": "0"}).
  33. 33 ExpectElementCount("#capacity-error", 1)
  34. 34 })
  35. 35
  36. 36 specs.Add("Visitor creates a workshop", func(bld *spec.B) {
  37. 37 created := evt.WorkshopCreated{Title: "Executable Specs", Capacity: 2}
  38. 38 bld.When(spec.Post("/workshops", spec.Form{"title": " Executable Specs ", "capacity": "2"})).
  39. 39 Then("creation is recorded and the visitor receives its shareable destination").
  40. 40 Covers("W1").
  41. 41 ExpectEvents(created).
  42. 42 Expect(createdWorkshopLink(created))
  43. 43 })
  44. 44
  45. 45 for _, example := range []struct {
  46. 46 name string
  47. 47 fields spec.Form
  48. 48 }{
  49. 49 {"blank title", spec.Form{"title": " ", "capacity": "2"}},
  50. 50 {"zero capacity", spec.Form{"title": "Specs", "capacity": "0"}},
  51. 51 {"negative capacity", spec.Form{"title": "Specs", "capacity": "-1"}},
  52. 52 {"fractional capacity", spec.Form{"title": "Specs", "capacity": "1.5"}},
  53. 53 {"missing capacity", spec.Form{"title": "Specs"}},
  54. 54 } {
  55. 55 specs.Add("Visitor submits invalid workshop creation: "+example.name, func(bld *spec.B) {
  56. 56 bld.When(spec.Post("/workshops", example.fields)).
  57. 57 Then("invalid input creates no workshop").
  58. 58 Covers("W1").
  59. 59 ExpectStatus(http.StatusBadRequest).
  60. 60 ExpectNoEvents()
  61. 61 })
  62. 62 }
  63. 63
  64. 64 return specs
  65. 65}

  67. 67func createdWorkshopLink(expected evt.WorkshopCreated) spec.Expectation {
  68. 68 return spec.Custom(func(result spec.Result) error {
  69. 69 location := result.Header.Get("Location")
  70. 70 id, ok := strings.CutPrefix(location, "/workshops/")
  71. 71 if !ok || id == "" {
  72. 72 return fmt.Errorf("creation needs a workshop destination with a nonempty ID; got %q", location)
  73. 73 }
  74. 74 created := expected
  75. 75 created.WorkshopID = id
  76. 76 return errors.Join(spec.Verify(
  77. 77 []spec.Expectation{spec.Redirect(location), spec.Events(created)},
  78. 78 result,
  79. 79 )...)
  80. 80 })
