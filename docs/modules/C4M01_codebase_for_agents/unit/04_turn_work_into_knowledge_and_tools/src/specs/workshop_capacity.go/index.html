  9. 9func workshopCapacitySpecs() spec.Specs {
  10. 10 var specs spec.Specs
  11. 11 workshop := evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 3}
  12. 12 alice := evt.SeatReserved{WorkshopID: "october", Email: "alice@example.com"}
  13. 13 bob := evt.SeatReserved{WorkshopID: "october", Email: "bob@example.com"}
  14. 14
  15. 15 specs.Add("Visitor opens the capacity form", func(bld *spec.B) {
  16. 16 bld.Given(workshop, alice, bob)
  17. 17 bld.When(spec.Get("/workshops/october/reservations")).
  18. 18 Then("the reservations page offers the current total for editing").
  19. 19 Covers("W15", "W11").
  20. 20 ExpectEl(`form:has(input[name="capacity"])`, spec.With{"method": "post", "action": "/workshops/october/capacity"}).
  21. 21 ExpectEl(`form input[name="capacity"]`, spec.With{"value": "3"})
  22. 22 })
  23. 23
  24. 24 specs.Add("Visitor increases workshop capacity", func(bld *spec.B) {
  25. 25 bld.Given(workshop, alice)
  26. 26 bld.When(spec.Post("/workshops/october/capacity", spec.Form{"capacity": "4"})).
  27. 27 Then("the new total is recorded and the visitor returns to reservations").
  28. 28 Covers("W15").
  29. 29 ExpectRedirect("/workshops/october/reservations").
  30. 30 ExpectEvents(evt.WorkshopCapacityChanged{WorkshopID: "october", Capacity: 4})
  31. 31 })
  32. 32
  33. 33 specs.Add("Visitor reduces capacity to the number of reservations", func(bld *spec.B) {
  34. 34 bld.Given(workshop, alice, bob)
  35. 35 bld.When(spec.Post("/workshops/october/capacity", spec.Form{"capacity": "2"})).
  36. 36 Then("the workshop can become full without removing reservations").
  37. 37 Covers("W15").
  38. 38 ExpectRedirect("/workshops/october/reservations").
  39. 39 ExpectEvents(evt.WorkshopCapacityChanged{WorkshopID: "october", Capacity: 2})
  40. 40 })
  41. 41
  42. 42 specs.Add("Visitor submits the current capacity", func(bld *spec.B) {
  43. 43 bld.Given(workshop, alice)
  44. 44 bld.When(spec.Post("/workshops/october/capacity", spec.Form{"capacity": "3"})).
  45. 45 Then("an unchanged total needs no new event").
  46. 46 Covers("W15").
  47. 47 ExpectRedirect("/workshops/october/reservations").
  48. 48 ExpectNoEvents()
  49. 49 })
  50. 50
  51. 51 specs.Add("Visitor requests fewer seats than are reserved", func(bld *spec.B) {
  52. 52 bld.Given(workshop, alice, bob)
  53. 53 bld.When(spec.Post("/workshops/october/capacity", spec.Form{"capacity": "1"})).
  54. 54 Then("existing reservations set the minimum and the visitor can correct the request").
  55. 55 Covers("W15", "W12").
  56. 56 ExpectStatus(http.StatusConflict).
  57. 57 ExpectNoEvents().
  58. 58 ExpectEl(`form input[name="capacity"]`, spec.With{"value": "1"}).
  59. 59 ExpectEl("#capacity-error")
  60. 60 })
  61. 61
  62. 62 for _, example := range []struct {
  63. 63 name string
  64. 64 fields spec.Form
  65. 65 value string
  66. 66 }{
  67. 67 {"missing capacity", spec.Form{}, ""},
  68. 68 {"zero capacity", spec.Form{"capacity": "0"}, "0"},
  69. 69 {"negative capacity", spec.Form{"capacity": "-1"}, "-1"},
  70. 70 {"fractional capacity", spec.Form{"capacity": "1.5"}, "1.5"},
  71. 71 } {
  72. 72 specs.Add("Visitor submits invalid capacity: "+example.name, func(bld *spec.B) {
  73. 73 bld.Given(workshop, alice)
  74. 74 bld.When(spec.Post("/workshops/october/capacity", example.fields)).
  75. 75 Then("invalid capacity records nothing and can be corrected in place").
  76. 76 Covers("W15", "W12").
  77. 77 ExpectStatus(http.StatusBadRequest).
  78. 78 ExpectNoEvents().
  79. 79 ExpectEl(`form input[name="capacity"]`, spec.With{"value": example.value}).
  80. 80 ExpectEl("#capacity-error")
  81. 81 })
  82. 82 }
  83. 83
  84. 84 changed := evt.WorkshopCapacityChanged{WorkshopID: "october", Capacity: 4}
  85. 85 specs.Add("Organizer reads occupancy after a capacity change", func(bld *spec.B) {
  86. 86 bld.Given(workshop, alice, bob, evt.WorkshopCapacityChanged{WorkshopID: "october", Capacity: 2})
  87. 87 bld.When(spec.Get("/workshops/october/reservations")).
  88. 88 Then("reducing the total to occupancy keeps both reservations and makes the workshop full").
  89. 89 Covers("W15", "W11").
  90. 90 ExpectEl("#reserved", spec.With{"text": "2"}).
  91. 91 ExpectEl("#capacity", spec.With{"text": "2"}).
  92. 92 ExpectEl("#remaining", spec.With{"text": "0"}).
  93. 93 ExpectTexts("[data-reservation]", "alice@example.com", "bob@example.com")
  94. 94 })
  95. 95
  96. 96 specs.Add("Organizer reads the catalogue after a capacity change", func(bld *spec.B) {
  97. 97 bld.Given(workshop, alice, changed)
  98. 98 bld.When(spec.Get("/")).
  99. 99 Then("the new total determines visible availability").
  100. 100 Covers("W15", "W8").
  101. 101 ExpectVisibleText(workshopCatalogueRow("october"), "1 / 4 reserved", "3 remaining", "Available")
  102. 102 })
  103. 103
  104. 104 specs.Add("Participant checks availability after a capacity increase", func(bld *spec.B) {
  105. 105 bld.Given(
  106. 106 evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 1},
  107. 107 alice,
  108. 108 evt.WorkshopCapacityChanged{WorkshopID: "october", Capacity: 2},
  109. 109 )
  110. 110 bld.When(spec.Get("/workshops/october?email=bob%40example.com")).
  111. 111 Then("an added seat is available to another participant").
  112. 112 Covers("W15", "W2", "W3").
  113. 113 ExpectEl("#capacity", spec.With{"text": "2"}).
  114. 114 ExpectEl("#remaining", spec.With{"text": "1"}).
  115. 115 ExpectEl("#reserve-form", spec.With{"action": "/workshops/october/reserve"})
  116. 116 })
  117. 117
  118. 118 return specs
  119. 119}
