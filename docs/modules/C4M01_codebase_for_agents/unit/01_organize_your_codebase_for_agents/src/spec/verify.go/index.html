  38. 38// Expectation is structured evidence for a spec's promise.
  39. 39type Expectation struct {
  40. 40 kind expectationKind
  41. 41 status int
  42. 42 uri string
  43. 43 events []evt.Event
  44. 44 selector string
  45. 45 count int
  46. 46 text string
  47. 47 texts []string
  48. 48 fragments []string
  49. 49 properties With
  50. 50 custom func(Result) error
  51. 51}
  52. 52
  53. 53// With describes one element: "text" is normalized text, "text.exact" is raw
  54. 54// text, and other keys name attributes. Unspecified properties are ignored.
  55. 55type With map[string]string

  79. 79func El(selector string, expected ...With) Expectation {
  80. 80 if len(expected) > 1 {
  81. 81 panic("element expectation accepts at most one property map")
  82. 82 }
  83. 83 if len(expected) == 1 && len(expected[0]) == 0 {
  84. 84 panic("element expectation property map must not be empty")
  85. 85 }
  86. 86 var want With
  87. 87 if len(expected) == 1 {
  88. 88 want = make(With, len(expected[0]))
  89. 89 for key, value := range expected[0] {
  90. 90 if strings.TrimSpace(key) == "" {
  91. 91 panic("HTML expectation key must not be empty")
  92. 92 }
  93. 93 want[key] = value
  94. 94 }
  95. 95 }
  96. 96 return Expectation{kind: expectElement, selector: selector, properties: want}
  97. 97}

  109. 109func VisibleText(selector string, fragments ...string) Expectation {
  110. 110 if len(fragments) == 0 {
  111. 111 panic("visible text expectation requires at least one fragment")
  112. 112 }
  113. 113 normalized := make([]string, len(fragments))
  114. 114 for index, fragment := range fragments {
  115. 115 normalized[index] = normalizeText(fragment)
  116. 116 if normalized[index] == "" {
  117. 117 panic("visible text fragment must not be empty")
  118. 118 }
  119. 119 }
  120. 120 return Expectation{kind: expectVisibleText, selector: selector, fragments: normalized}

  132. 132func Verify(expectations []Expectation, result Result) []error {
  133. 133 var document *goquery.Document
  134. 134 needsDocument := false
  135. 135 for _, expectation := range expectations {
  136. 136 switch expectation.kind {
  137. 137 case expectCount, expectText, expectElement, expectTexts, expectNoText, expectVisibleText:
  138. 138 needsDocument = true
  139. 139 }
  140. 140 }
  141. 141 if needsDocument {
  142. 142 var err error
  143. 143 document, err = goquery.NewDocumentFromReader(strings.NewReader(string(result.Body)))
  144. 144 if err != nil {
  145. 145 return []error{fmt.Errorf("read response HTML: %w", err)}
  146. 146 }
  147. 147 }
  148. 148
  149. 149 var failures []error
  150. 150 for _, expectation := range expectations {
  151. 151 if err := verify(expectation, result, document); err != nil {
  152. 152 failures = append(failures, err)
  153. 153 }
  154. 154 }
  155. 155 return failures
  156. 156}

  158. 158func verify(expectation Expectation, result Result, document *goquery.Document) error {
  159. 159 switch expectation.kind {
  160. 160 case expectStatus:
  161. 161 return verifyStatus(expectation.status, result)
  162. 162 case expectRedirect:
  163. 163 err := verifyStatus(http.StatusSeeOther, result)
  164. 164 if got := result.Header.Get("Location"); got != expectation.uri {
  165. 165 err = errors.Join(err, difference("redirect location", fmt.Sprintf("%q", expectation.uri), fmt.Sprintf("%q", got)))
  166. 166 }
  167. 167 return err
  168. 168 case expectEvents:
  169. 169 return verifyEvents(expectation.events, result.Events)
  170. 170 case expectCount:
  171. 171 return verifyCount(document, expectation.selector, expectation.count)
  172. 172 case expectText:
  173. 173 if err := verifyCount(document, expectation.selector, 1); err != nil {
  174. 174 return err
  175. 175 }
  176. 176 if got := normalizeText(document.Find(expectation.selector).Text()); got != expectation.text {
  177. 177 return difference("text: "+expectation.selector, fmt.Sprintf("%q", expectation.text), fmt.Sprintf("%q", got))
  178. 178 }
  179. 179 return nil
  180. 180 case expectElement:
  181. 181 return verifyElement(document, expectation.selector, expectation.properties)
  182. 182 case expectTexts:
  183. 183 var got []string
  184. 184 document.Find(expectation.selector).Each(func(_ int, element *goquery.Selection) {
  185. 185 got = append(got, normalizeText(element.Text()))
  186. 186 })
  187. 187 slices.Sort(got)
  188. 188 if !slices.Equal(got, expectation.texts) {
  189. 189 return difference("texts: "+expectation.selector, fmt.Sprintf("%q", expectation.texts), fmt.Sprintf("%q", got))
  190. 190 }
  191. 191 return nil
  192. 192 case expectNoText:
  193. 193 if strings.Contains(document.Text(), expectation.text) {
  194. 194 return fmt.Errorf("page unexpectedly contains %q", expectation.text)
  195. 195 }
  196. 196 return nil
  197. 197 case expectVisibleText:
  198. 198 if err := verifyCount(document, expectation.selector, 1); err != nil {
  199. 199 return err
  200. 200 }
  201. 201 actual := normalizeText(document.Find(expectation.selector).Text())
  202. 202 var missing []string
  203. 203 for _, fragment := range expectation.fragments {
  204. 204 if !strings.Contains(actual, fragment) {
  205. 205 missing = append(missing, fragment)
  206. 206 }
  207. 207 }
  208. 208 if len(missing) > 0 {
  209. 209 return fmt.Errorf(
  210. 210 "visible text: %s\n missing: %q\n got: %q",
  211. 211 expectation.selector,
  212. 212 missing,
  213. 213 actual,
  214. 214 )
  215. 215 }
  216. 216 return nil
  217. 217 case expectCustom:
  218. 218 return expectation.custom(result)
  219. 219 default:
  220. 220 return fmt.Errorf("unknown expectation kind %d", expectation.kind)
  221. 221 }
  222. 222}

  224. 224func verifyStatus(want int, result Result) error {
  225. 225 if result.StatusCode != want {
  226. 226 return difference("HTTP status", fmt.Sprintf("%d %s", want, http.StatusText(want)), fmt.Sprintf("%d %s", result.StatusCode, http.StatusText(result.StatusCode)))
  227. 227 }
  228. 228 return nil
  229. 229}

  231. 231func verifyEvents(want, got []evt.Event) error {
  232. 232 if len(want) == 0 {
  233. 233 if len(got) != 0 {
  234. 234 return difference("recorded events", DescribeEvents(want), DescribeEvents(got))
  235. 235 }
  236. 236 return nil
  237. 237 }
  238. 238 actualIndex := 0
  239. 239 for index, expected := range want {
  240. 240 matched := false
  241. 241 var mismatch error
  242. 242 for actualIndex < len(got) {
  243. 243 actual := got[actualIndex]
  244. 244 actualIndex++
  245. 245 if evt.Name(expected) != evt.Name(actual) {
  246. 246 continue
  247. 247 }
  248. 248 if err := compareFields(fmt.Sprintf("event %d %T", index+1, expected), expected, actual); err != nil {
  249. 249 if mismatch == nil {
  250. 250 mismatch = err
  251. 251 }
  252. 252 continue
  253. 253 }
  254. 254 matched = true
  255. 255 break
  256. 256 }
  257. 257 if !matched {
  258. 258 if mismatch != nil {
  259. 259 return mismatch
  260. 260 }
  261. 261 return difference(fmt.Sprintf("event %d was not found in order", index+1), DescribeEvents([]evt.Event{expected}), DescribeEvents(got))
  262. 262 }
  263. 263 }
  264. 264 return nil
  265. 265}

  267. 267func verifyCount(document *goquery.Document, selector string, want int) error {
  268. 268 if got := document.Find(selector).Length(); got != want {
  269. 269 return difference("element count: "+selector, want, got)
  270. 270 }
  271. 271 return nil
  272. 272}

  274. 274func verifyElement(document *goquery.Document, selector string, want With) error {
  275. 275 selection := document.Find(selector)
  276. 276 if want == nil {
  277. 277 if selection.Length() == 0 {
  278. 278 return fmt.Errorf("HTML selector %q matched no elements", selector)
  279. 279 }
  280. 280 return nil
  281. 281 }
  282. 282 if err := verifyCount(document, selector, 1); err != nil {
  283. 283 return err
  284. 284 }
  285. 285 actual := With{}
  286. 286 for _, attribute := range selection.Get(0).Attr {
  287. 287 actual[attribute.Key] = attribute.Val
  288. 288 }
  289. 289 actual["text"] = normalizeText(selection.Text())
  290. 290 actual["text.exact"] = selection.Text()
  291. 291 return compareFields(fmt.Sprintf("HTML selector %q", selector), want, actual)
  292. 292}

  294. 294func difference(subject string, want, got any) error {
  295. 295 return fmt.Errorf("%s\n want: %v\n got: %v", subject, want, got)
  296. 296}

  298. 298func compareFields(subject string, want, got any) error {
  299. 299 expected, err := left.FlattenJSONValue(want)
  300. 300 if err != nil {
  301. 301 return fmt.Errorf("%s: prepare expected fields: %w", subject, err)
  302. 302 }
  303. 303 actual, err := left.FlattenJSONBytes([]byte(jsonText(got)))
  304. 304 if err != nil {
  305. 305 return fmt.Errorf("%s: read actual fields: %w", subject, err)
  306. 306 }
  307. 307 if deltas := left.LeftMapDeltas(expected, actual); len(deltas) > 0 {
  308. 308 return fmt.Errorf("%s:\n%s", subject, strings.Join(deltas, "\n"))
  309. 309 }
  310. 310 return nil
  311. 311}

  313. 313func jsonText(value any) string {
  314. 314 payload, err := json.Marshal(value)
  315. 315 if err != nil {
  316. 316 panic(err)
  317. 317 }
  318. 318 return string(payload)
  319. 319}

  321. 321func DescribeEvents(events []evt.Event) string {
  322. 322 if len(events) == 0 {
  323. 323 return "no events"
  324. 324 }
  325. 325 descriptions := make([]string, 0, len(events))
  326. 326 for _, event := range events {
  327. 327 descriptions = append(descriptions, fmt.Sprintf("%T %s", event, jsonText(event)))
  328. 328 }
  329. 329 return strings.Join(descriptions, "; ")
  330. 330}

  332. 332func normalizeText(value string) string {
  333. 333 return strings.Join(strings.Fields(value), " ")
