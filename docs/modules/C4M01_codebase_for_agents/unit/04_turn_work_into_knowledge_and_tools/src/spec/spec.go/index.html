  52. 52func ParseRequirementID(value string) (RequirementID, error) {

  103. 103func (b *StepBuilder) Covers(requirements ...string) *StepBuilder {
  104. 104 if len(b.spec.Requirements) != 0 {
  105. 105 panic("scenario requirements are already set")
  106. 106 }
  107. 107 if len(requirements) == 0 {
  108. 108 panic("Covers needs at least one requirement")
  109. 109 }
  110. 110 seen := make(map[RequirementID]bool, len(requirements))
  111. 111 for _, value := range requirements {
  112. 112 id, err := ParseRequirementID(value)
  113. 113 if err != nil {
  114. 114 panic(err)

  116. 116 if seen[id] {
  117. 117 panic(fmt.Sprintf("duplicate requirement ID %q", id))
  118. 118 }
  119. 119 seen[id] = true
  120. 120 b.spec.Requirements = append(b.spec.Requirements, id)
  121. 121 }
  122. 122 return b
  123. 123}
