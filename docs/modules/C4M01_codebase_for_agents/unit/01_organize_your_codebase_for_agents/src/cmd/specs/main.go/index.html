  23. 23func run(args []string, out io.Writer, features []spec.Feature, contractPath string) int {
  24. 24 flags := flag.NewFlagSet("specs", flag.ContinueOnError)
  25. 25 flags.SetOutput(out)
  26. 26 coverage := flags.Bool("coverage", false, "report requirements, linked scenarios, and their outcomes after the full run")
  27. 27 if err := flags.Parse(args); err != nil {
  28. 28 if errors.Is(err, flag.ErrHelp) {
  29. 29 return 0
  30. 30 }
  31. 31 return 2
  32. 32 }
  33. 33 if flags.NArg() != 0 {
  34. 34 fmt.Fprintf(out, "specs: unexpected arguments: %s\n", strings.Join(flags.Args(), " "))
  35. 35 return 2
  36. 36 }
  37. 37 total := 0
  38. 38 for _, feature := range features {
  39. 39 total += len(feature.Specs)
  40. 40 }
  41. 41 if total == 0 {
  42. 42 fmt.Fprintln(out, "FAIL No scenarios to run.")
  43. 43 return 1
  44. 44 }
  45. 45 contract, err := os.Open(contractPath)
  46. 46 if err != nil {
  47. 47 fmt.Fprintf(out, "specs: requirements: %v\n", err)
  48. 48 return 2
  49. 49 }
  50. 50 requirements, err := readRequirements(contract)
  51. 51 _ = contract.Close()
  52. 52 if err == nil {
  53. 53 err = validateRequirements(requirements, features)
  54. 54 }
  55. 55 if err != nil {
  56. 56 fmt.Fprintf(out, "specs: %s: %v\n", contractPath, err)
  57. 57 return 2
  58. 58 }
  59. 59 failed := 0
  60. 60 expectations := 0
  61. 61 var outcomes []scenarioOutcome
  62. 62 for _, feature := range features {
  63. 63 for _, scenario := range feature.Specs {
  64. 64 expectations += scenario.ExpectationCount()
  65. 65 failures := execute(scenario)
  66. 66 outcomes = append(outcomes, scenarioOutcome{Feature: feature.Name, Scenario: scenario, Failures: failures})
  67. 67 if len(failures) == 0 {
  68. 68 continue
  69. 69 }
  70. 70 failed++
  71. 71 fmt.Fprintf(out, "FAIL %s · %s / %s\n", describeRequirements(scenario.Requirements), feature.Name, scenario.Name)
  72. 72 fmt.Fprintf(out, " Then %s\n", scenario.Meaning)
  73. 73 fmt.Fprintf(out, " Given %s\n", spec.DescribeEvents(scenario.GivenEvents))
  74. 74 fmt.Fprintf(out, " When %s %s\n", scenario.Request.Method, scenario.Request.URI)
  75. 75 if scenario.Request.Form != nil {
  76. 76 fmt.Fprintf(out, " Form %s\n", jsonText(scenario.Request.Form))
  77. 77 }
  78. 78 for _, err := range failures {
  79. 79 fmt.Fprintf(out, " %s\n", strings.ReplaceAll(err.Error(), "\n", "\n "))
  80. 80 }
  81. 81 fmt.Fprintln(out)
  82. 82 }
  83. 83 }
  84. 84 fmt.Fprintf(out, "%d scenarios: %d passed, %d failed · %d expectations\n", total, total-failed, failed, expectations)
  85. 85 if *coverage {
  86. 86 printCoverage(out, requirements, outcomes)
  87. 87 }
  88. 88 if failed > 0 {
  89. 89 return 1
  90. 90 }
  91. 91 return 0
  92. 92}

  94. 94func execute(scenario spec.Spec) (failures []error) {
  95. 95 defer func() {
  96. 96 if value := recover(); value != nil {
  97. 97 failures = append(failures, fmt.Errorf("scenario panicked: %v", value))
  98. 98 }
  99. 99 }()
  100. 100 if scenario.WhenCount != 1 || scenario.Request.Method == "" || scenario.Request.URI == "" {
  101. 101 return []error{fmt.Errorf("a scenario needs exactly one When request")}
  102. 102 }
  103. 103 if scenario.Meaning == "" || len(scenario.Expectations) == 0 {
  104. 104 return []error{fmt.Errorf("a scenario needs a Then promise and at least one expectation")}
  105. 105 }
  106. 106 store, err := app.Open(":memory:")
  107. 107 if err != nil {
  108. 108 return []error{fmt.Errorf("open scenario database: %w", err)}
  109. 109 }
  110. 110 defer func() {
  111. 111 if err := store.Close(); err != nil {
  112. 112 failures = append(failures, fmt.Errorf("close scenario database: %w", err))
  113. 113 }
  114. 114 }()
  115. 115 ctx := context.Background()
  116. 116 if err := store.Transact(ctx, func(tx *app.Tx) error { return tx.Emit(scenario.GivenEvents...) }); err != nil {
  117. 117 return []error{fmt.Errorf("Given events: %w", err)}
  118. 118 }
  119. 119 before, err := store.History(ctx)
  120. 120 if err != nil {
  121. 121 return []error{fmt.Errorf("read history before request: %w", err)}
  122. 122 }
  123. 123 req := httptest.NewRequest(scenario.Request.Method, scenario.Request.URI, strings.NewReader(scenario.Request.Form.Encode()))
  124. 124 if scenario.Request.Form != nil {
  125. 125 req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
  126. 126 }
  127. 127 response := httptest.NewRecorder()
  128. 128 app.Handler(store).ServeHTTP(response, req)
  129. 129 after, err := store.History(ctx)
  130. 130 if err != nil {
  131. 131 return []error{fmt.Errorf("read history after request: %w", err)}
  132. 132 }
  133. 133 if len(after) < len(before) {
  134. 134 return []error{fmt.Errorf("request removed existing event history")}
  135. 135 }
  136. 136 for index, event := range before {
  137. 137 if !reflect.DeepEqual(event, after[index]) {
  138. 138 return []error{fmt.Errorf("request changed historical event %d", index+1)}
  139. 139 }
  140. 140 }
  141. 141 return spec.Verify(scenario.Expectations, spec.Result{
  142. 142 StatusCode: response.Code,
  143. 143 Header: response.Header(),
  144. 144 Body: response.Body.Bytes(),
  145. 145 Events: after[len(before):],
  146. 146 })
  147. 147}
