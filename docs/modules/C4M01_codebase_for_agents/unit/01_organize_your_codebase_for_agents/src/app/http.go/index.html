  22. 22func Handler(store *Store) http.Handler {
  23. 23 mux := http.NewServeMux()
  24. 24 mux.HandleFunc("GET /{$}", handle(store, home))
  25. 25 mux.HandleFunc("GET /workshops/new", handle(store, showCreation))
  26. 26 mux.HandleFunc("POST /workshops", handle(store, createWorkshop))
  27. 27 mux.HandleFunc("GET /workshops/{id}", handle(store, showWorkshop))
  28. 28 mux.HandleFunc("GET /workshops/{id}/reservations", handle(store, showReservations))
  29. 29 mux.HandleFunc("POST /workshops/{id}/reserve", handle(store, reserveSeat))
  30. 30 mux.HandleFunc("GET /style.css", func(w http.ResponseWriter, r *http.Request) {
  31. 31 w.Header().Set("Content-Type", "text/css; charset=utf-8")
  32. 32 body, _ := assets.ReadFile("style.css")
  33. 33 _, _ = w.Write(body)

  38. 38func handle(store *Store, action func(*Tx, *http.Request) (response, error)) http.HandlerFunc {
  39. 39 return func(w http.ResponseWriter, r *http.Request) {
  40. 40 r.Body = http.MaxBytesReader(w, r.Body, 64<<10)
  41. 41 var result response
  42. 42 err := store.Transact(r.Context(), func(tx *Tx) error {
  43. 43 var err error
  44. 44 result, err = action(tx, r)
  45. 45 return err
  46. 46 })
  47. 47 if err != nil {
  48. 48 log.Printf("%s %s: %v", r.Method, r.URL.Path, err)
  49. 49 http.Error(w, "The request could not be completed.", http.StatusInternalServerError)
  50. 50 return
  51. 51 }
  52. 52 if result.location != "" {
  53. 53 http.Redirect(w, r, result.location, result.status)
  54. 54 return
  55. 55 }
  56. 56 w.Header().Set("Content-Type", "text/html; charset=utf-8")
  57. 57 w.WriteHeader(result.status)
  58. 58 _, _ = w.Write(result.body)
  59. 59 }
  60. 60}

  74. 74func createWorkshop(tx *Tx, r *http.Request) (response, error) {
  75. 75 if err := r.ParseForm(); err != nil {
  76. 76 return render("new_workshop", "New workshop", creationForm{FormError: "The form could not be read. Please try again."}, http.StatusBadRequest)
  77. 77 }
  78. 78 form := creationForm{Title: r.PostForm.Get("title"), Capacity: r.PostForm.Get("capacity")}
  79. 79 title := strings.TrimSpace(form.Title)
  80. 80 capacity, err := strconv.Atoi(form.Capacity)
  81. 81 if title == "" {
  82. 82 form.TitleError = "Enter a workshop title."
  83. 83 }
  84. 84 if err != nil || capacity < 1 {
  85. 85 form.CapacityError = "Enter a positive whole number."
  86. 86 }
  87. 87 if form.TitleError != "" || form.CapacityError != "" {
  88. 88 return render("new_workshop", "New workshop", form, http.StatusBadRequest)
  89. 89 }
  90. 90 id := rand.Text()
  91. 91 if err := tx.Emit(evt.WorkshopCreated{WorkshopID: id, Title: title, Capacity: capacity}); err != nil {
  92. 92 return response{}, err
  93. 93 }
  94. 94 return response{status: http.StatusSeeOther, location: "/workshops/" + id}, nil
  95. 95}

  97. 97func normalizeEmail(raw string) (string, error) {
  98. 98 email := strings.ToLower(strings.TrimSpace(raw))
  99. 99 address, err := mail.ParseAddress(email)
  100. 100 if err != nil || address.Address != email || address.Name != "" {
  101. 101 return "", errors.New("enter one email address without a display name")
  102. 102 }
  103. 103 return email, nil
  104. 104}

  106. 106func showWorkshop(tx *Tx, r *http.Request) (response, error) {
  107. 107 workshop, err := tx.workshop(r.PathValue("id"))
  108. 108 if errors.Is(err, sql.ErrNoRows) {
  109. 109 return problem(http.StatusNotFound, "Workshop not found.")
  110. 110 }
  111. 111 if err != nil {
  112. 112 return response{}, err
  113. 113 }
  114. 114 page := workshopPage{Workshop: workshop}
  115. 115 if raw := r.URL.Query().Get("email"); raw != "" {
  116. 116 page.Email, err = normalizeEmail(raw)
  117. 117 if err != nil {
  118. 118 page.Email, page.EmailError = raw, "Enter one valid email address without a display name."
  119. 119 return render("workshop", workshop.Title, page, http.StatusBadRequest)
  120. 120 }
  121. 121 page.Reserved, err = tx.isReserved(workshop.ID, page.Email)
  122. 122 if err != nil {
  123. 123 return response{}, err
  124. 124 }
  125. 125 }
  126. 126 return render("workshop", workshop.Title, page, http.StatusOK)
  127. 127}

  144. 144func reserveSeat(tx *Tx, r *http.Request) (response, error) {
  145. 145 workshop, err := tx.workshop(r.PathValue("id"))
  146. 146 if errors.Is(err, sql.ErrNoRows) {
  147. 147 return problem(http.StatusNotFound, "Workshop not found.")
  148. 148 }
  149. 149 if err != nil {
  150. 150 return response{}, err
  151. 151 }
  152. 152 if err := r.ParseForm(); err != nil {
  153. 153 return render("workshop", workshop.Title, workshopPage{Workshop: workshop, EmailError: "The form could not be read. Enter your email again."}, http.StatusBadRequest)
  154. 154 }
  155. 155 email, err := normalizeEmail(r.PostForm.Get("email"))
  156. 156 if err != nil {
  157. 157 return render("workshop", workshop.Title, workshopPage{
  158. 158 Workshop: workshop, Email: r.PostForm.Get("email"), EmailError: "Enter one valid email address without a display name.",
  159. 159 }, http.StatusBadRequest)
  160. 160 }
  161. 161 reserved, err := tx.isReserved(workshop.ID, email)
  162. 162 if err != nil {
  163. 163 return response{}, err
  164. 164 }
  165. 165 // Check identity before capacity: retrying an existing reservation succeeds
  166. 166 // even if that reservation took the final seat.
  167. 167 if !reserved {
  168. 168 if workshop.Remaining == 0 {
  169. 169 return render("workshop", workshop.Title, workshopPage{Workshop: workshop, Email: email}, http.StatusConflict)
  170. 170 }
  171. 171 if err := tx.Emit(evt.SeatReserved{WorkshopID: workshop.ID, Email: email}); err != nil {
  172. 172 return response{}, err
  173. 173 }
  174. 174 }
  175. 175 return response{
  176. 176 status: http.StatusSeeOther,
  177. 177 location: "/workshops/" + workshop.ID + "?" + url.Values{"email": {email}}.Encode(),
  178. 178 }, nil
