  9. 9func releaseSeatSpecs() spec.Specs {
  10. 10 var specs spec.Specs
  11. 11 workshop := evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 1}
  12. 12 aliceReserved := evt.SeatReserved{WorkshopID: "october", Email: "alice@example.com"}
  13. 13 aliceReleased := evt.SeatReleased{WorkshopID: "october", Email: "alice@example.com"}
  14. 14
  15. 15 specs.Add("Participant checks a reserved email to release a seat", func(bld *spec.B) {
  16. 16 bld.Given(workshop, aliceReserved)
  17. 17 bld.When(spec.Get("/workshops/october?email=ALICE%40EXAMPLE.COM")).
  18. 18 Then("a reserved participant can release the seat even when the workshop is full").
  19. 19 Covers("W13", "W2", "W3").
  20. 20 ExpectEl("#reservation-status", spec.With{"data-status": "reserved"}).
  21. 21 ExpectEl("#release-form", spec.With{"method": "post", "action": "/workshops/october/release"}).
  22. 22 ExpectEl(`#release-form input[name="email"]`, spec.With{"value": "alice@example.com"}).
  23. 23 ExpectEl(`#release-form button[type="submit"]`).
  24. 24 ExpectNoElement("#reserve-form").
  25. 25 ExpectNoElement(`a[href]:not([href^="#"])`)
  26. 26 })
  27. 27
  28. 28 specs.Add("Participant releases a reserved seat", func(bld *spec.B) {
  29. 29 bld.Given(workshop, aliceReserved)
  30. 30 bld.When(spec.Post("/workshops/october/release", spec.Form{"email": " Alice@Example.COM "})).
  31. 31 Then("the release is recorded for the normalized email and redirects to its status").
  32. 32 Covers("W13", "W3").
  33. 33 ExpectRedirect("/workshops/october?email=alice%40example.com").
  34. 34 ExpectEvents(aliceReleased)
  35. 35 })
  36. 36
  37. 37 for _, example := range []struct {
  38. 38 name string
  39. 39 history []evt.Event
  40. 40 }{
  41. 41 {"without a reservation", []evt.Event{workshop}},
  42. 42 {"after already releasing", []evt.Event{workshop, aliceReserved, aliceReleased}},
  43. 43 {"while another email holds the last seat", []evt.Event{workshop, evt.SeatReserved{WorkshopID: "october", Email: "bob@example.com"}}},
  44. 44 } {
  45. 45 specs.Add("Participant releases "+example.name, func(bld *spec.B) {
  46. 46 bld.Given(example.history...)
  47. 47 bld.When(spec.Post("/workshops/october/release", spec.Form{"email": " ALICE@example.com "})).
  48. 48 Then("an unreserved email succeeds without recording a release").
  49. 49 Covers("W13").
  50. 50 ExpectRedirect("/workshops/october?email=alice%40example.com").
  51. 51 ExpectNoEvents()
  52. 52 })
  53. 53 }
  54. 54
  55. 55 for _, example := range []struct{ name, email string }{
  56. 56 {"missing email", ""},
  57. 57 {"whitespace email", " "},
  58. 58 {"malformed email", "not-an-email"},
  59. 59 {"email with a display name", "Alice <alice@example.com>"},
  60. 60 {"multiple email addresses", "alice@example.com,bob@example.com"},
  61. 61 } {
  62. 62 specs.Add("Participant submits an invalid release: "+example.name, func(bld *spec.B) {
  63. 63 bld.Given(workshop, aliceReserved)
  64. 64 bld.When(spec.Post("/workshops/october/release", spec.Form{"email": example.email})).
  65. 65 Then("invalid email releases nothing and can be corrected on the participant page").
  66. 66 Covers("W13", "W12", "W2").
  67. 67 ExpectStatus(http.StatusBadRequest).
  68. 68 ExpectNoEvents().
  69. 69 ExpectEl("#email", spec.With{"value": example.email}).
  70. 70 ExpectEl("#email-error").
  71. 71 ExpectEl("form:has(#email)", spec.With{"method": "get", "action": "/workshops/october"}).
  72. 72 ExpectNoElement("#reservation-status, #reserve-form, #release-form").
  73. 73 ExpectNoElement(`a[href]:not([href^="#"])`)
  74. 74 })
  75. 75 }
  76. 76
  77. 77 specs.Add("Participant releases a seat in an unknown workshop", func(bld *spec.B) {
  78. 78 bld.Given(workshop, aliceReserved)
  79. 79 bld.When(spec.Post("/workshops/missing/release", spec.Form{"email": "alice@example.com"})).
  80. 80 Then("an unknown workshop cannot receive a release").
  81. 81 Covers("W9", "W13").
  82. 82 ExpectStatus(http.StatusNotFound).
  83. 83 ExpectNoEvents()
  84. 84 })
  85. 85
  86. 86 for _, email := range []string{"alice", "bob"} {
  87. 87 specs.Add("Participant "+email+" checks availability after a release", func(bld *spec.B) {
  88. 88 bld.Given(workshop, aliceReserved, aliceReleased)
  89. 89 bld.When(spec.Get("/workshops/october?email="+email+"%40example.com")).
  90. 90 Then("the seat is available to the released email and other visitors").
  91. 91 Covers("W14", "W2").
  92. 92 ExpectEl("#remaining", spec.With{"text": "1"}).
  93. 93 ExpectEl("#reservation-status", spec.With{"data-status": "not-reserved"}).
  94. 94 ExpectEl("#reserve-form", spec.With{"method": "post", "action": "/workshops/october/reserve"}).
  95. 95 ExpectEl(`#reserve-form input[name="email"]`, spec.With{"value": email + "@example.com"}).
  96. 96 ExpectNoElement("#release-form, #workshop-full").
  97. 97 ExpectNoElement(`a[href]:not([href^="#"])`)
  98. 98 })
  99. 99 specs.Add("Participant "+email+" reserves the released seat", func(bld *spec.B) {
  100. 100 bld.Given(workshop, aliceReserved, aliceReleased)
  101. 101 bld.When(spec.Post("/workshops/october/reserve", spec.Form{"email": email + "@example.com"})).
  102. 102 Then("the released seat can be reserved again").
  103. 103 Covers("W14", "W4").
  104. 104 ExpectRedirect("/workshops/october?email=" + email + "%40example.com").
  105. 105 ExpectEvents(evt.SeatReserved{WorkshopID: "october", Email: email + "@example.com"})
  106. 106 })
  107. 107 }
  108. 108
  109. 109 specs.Add("Participant opens a workshop after a release without an email", func(bld *spec.B) {
  110. 110 bld.Given(workshop, aliceReserved, aliceReleased)
  111. 111 bld.When(spec.Get("/workshops/october")).
  112. 112 Then("release remains unavailable until a reserved email is checked").
  113. 113 Covers("W13", "W2").
  114. 114 ExpectNoElement("#reservation-status, #reserve-form, #release-form")
  115. 115 })
  116. 116
  117. 117 specs.Add("Organizer reads the catalogue after a release", func(bld *spec.B) {
  118. 118 bld.Given(workshop, aliceReserved, aliceReleased)
  119. 119 bld.When(spec.Get("/")).
  120. 120 Then("the formerly full workshop shows reduced occupancy and is available").
  121. 121 Covers("W14", "W8").
  122. 122 ExpectVisibleText(workshopCatalogueRow("october"), "0 / 1 reserved", "1 remaining", "Available").
  123. 123 ExpectNoText("Full")
  124. 124 })
  125. 125
  126. 126 specs.Add("Organizer reads occupancy after a release", func(bld *spec.B) {
  127. 127 bld.Given(workshop, aliceReserved, aliceReleased)
  128. 128 bld.When(spec.Get("/workshops/october/reservations")).
  129. 129 Then("the organizer sees the released seat in the current counts").
  130. 130 Covers("W14", "W11").
  131. 131 ExpectEl("#reserved", spec.With{"text": "0"}).
  132. 132 ExpectEl("#capacity", spec.With{"text": "1"}).
  133. 133 ExpectEl("#remaining", spec.With{"text": "1"})

  136. 136 specs.Add("Organizer reads reservations after one participant releases", func(bld *spec.B) {
  137. 137 bld.Given(
  138. 138 evt.WorkshopCreated{WorkshopID: "october", Title: "Executable Specs", Capacity: 2},
  139. 139 aliceReserved,
  140. 140 evt.SeatReserved{WorkshopID: "october", Email: "bob@example.com"},
  141. 141 aliceReleased,
  142. 142 )
  143. 143 bld.When(spec.Get("/workshops/october/reservations")).
  144. 144 Then("only the released email leaves the reservations list").
  145. 145 Covers("W14", "W11").
  146. 146 ExpectTexts("[data-reservation]", "bob@example.com")

  148. 148
  149. 149 specs.Add("Participant checks the same email in another workshop after a release", func(bld *spec.B) {
  150. 150 bld.Given(
  151. 151 workshop, aliceReserved,
  152. 152 evt.WorkshopCreated{WorkshopID: "november", Title: "Foundations", Capacity: 1},
  153. 153 evt.SeatReserved{WorkshopID: "november", Email: "alice@example.com"},
  154. 154 aliceReleased,
  155. 155 )
  156. 156 bld.When(spec.Get("/workshops/november?email=alice%40example.com")).
  157. 157 Then("releasing one workshop leaves the other reservation and occupancy unchanged").
  158. 158 Covers("W14", "W7").
  159. 159 ExpectEl("#reservation-status", spec.With{"data-status": "reserved"}).
  160. 160 ExpectEl("#remaining", spec.With{"text": "0"})
  161. 161 })
  162. 162
  163. 163 return specs
