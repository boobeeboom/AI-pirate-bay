  20. 20// FlattenJSONValue normalizes a JSON-serializable value and flattens it into dotted paths.
  21. 21// Expected structs omit zero-valued fields; maps retain explicit zeros and nulls.
  22. 22// Nested arrays use `[index]`; nested object keys use `.field` when identifier-safe and
  23. 23// `["raw-key"]` otherwise.
  24. 24func FlattenJSONValue(value any) (map[string]any, error) {
  25. 25 normalized, err := normalizeExpectedJSONValue(reflect.ValueOf(value))
  26. 26 if err != nil {
  27. 27 return nil, err
  28. 28 }
  29. 29 if normalized == nil {
  30. 30 normalized = map[string]any{}
  31. 31 }
  32. 32
  33. 33 body, err := json.Marshal(normalized)
  34. 34 if err != nil {
  35. 35 return nil, err
  36. 36 }
  37. 37 return FlattenJSONBytes(body)
  38. 38}

  40. 40// FlattenJSONBytes parses a JSON object and flattens it into dotted paths.
  41. 41func FlattenJSONBytes(body []byte) (map[string]any, error) {
  42. 42 var decoded any
  43. 43 if err := json.Unmarshal(body, &decoded); err != nil {
  44. 44 return nil, err
  45. 45 }
  46. 46
  47. 47 root, ok := decoded.(map[string]any)
  48. 48 if !ok {
  49. 49 return nil, fmt.Errorf("json root must be an object")
  50. 50 }
  51. 51
  52. 52 out := make(map[string]any)
  53. 53 flattenFlatObject("", root, out)
  54. 54 return out, nil
  55. 55}

  57. 57// LeftMapDeltas compares only keys populated on expected against actual.
  58. 58// Extra keys on actual are ignored. Flatten both sides with these helpers first.
  59. 59func LeftMapDeltas(expected, actual map[string]any) []string {
  60. 60 if expected == nil {
  61. 61 return nil
  62. 62 }
  63. 63
  64. 64 keys := make([]string, 0, len(expected))
  65. 65 for key := range expected {
  66. 66 keys = append(keys, key)
  67. 67 }
  68. 68 sort.Strings(keys)
  69. 69
  70. 70 var deltas []string
  71. 71 for _, key := range keys {
  72. 72 expectedValue := expected[key]
  73. 73 actualValue, ok := actual[key]
  74. 74 if !ok {
  75. 75 deltas = append(deltas, formatDelta(key, formatFlatValue(expectedValue), "<unset>"))
  76. 76 continue
  77. 77 }
  78. 78 if !reflect.DeepEqual(expectedValue, actualValue) {
  79. 79 deltas = append(deltas, formatDelta(
  80. 80 key,
  81. 81 formatFlatValue(expectedValue),
  82. 82 formatFlatValue(actualValue),
  83. 83 ))
  84. 84 }
  85. 85 }
  86. 86
  87. 87 return deltas
  88. 88}

  90. 90func flattenFlatObject(path string, obj map[string]any, out map[string]any) {
  91. 91 if len(obj) == 0 {
  92. 92 if path != "" {
  93. 93 out[path] = emptyFlatObject
  94. 94 }
  95. 95 return
  96. 96 }
  97. 97
  98. 98 keys := make([]string, 0, len(obj))
  99. 99 for key := range obj {
  100. 100 keys = append(keys, key)
  101. 101 }
  102. 102 sort.Strings(keys)
  103. 103
  104. 104 for _, key := range keys {
  105. 105 flattenFlatValue(joinFlatPath(path, key), obj[key], out)
  106. 106 }
  107. 107}

  109. 109func flattenFlatValue(path string, value any, out map[string]any) {
  110. 110 switch typed := value.(type) {
  111. 111 case map[string]any:
  112. 112 flattenFlatObject(path, typed, out)
  113. 113 case []any:
  114. 114 if len(typed) == 0 {
  115. 115 out[path] = emptyFlatArray
  116. 116 return
  117. 117 }
  118. 118 for i, item := range typed {
  119. 119 flattenFlatValue(fmt.Sprintf("%s[%d]", path, i), item, out)
  120. 120 }
  121. 121 default:
  122. 122 out[path] = typed
  123. 123 }
  124. 124}

  126. 126func joinFlatPath(path, key string) string {
  127. 127 segment := formatFlatPathSegment(key)
  128. 128 if path == "" {
  129. 129 return segment
  130. 130 }
  131. 131 if segment != "" && segment[0] == '[' {
  132. 132 return path + segment
  133. 133 }
  134. 134 return path + "." + segment
  135. 135}

  144. 144func formatFlatValue(value any) string {
  145. 145 switch typed := value.(type) {
  146. 146 case emptyFlatContainer:
  147. 147 return string(typed)
  148. 148 default:

  157. 157func normalizeExpectedJSONValue(value reflect.Value) (any, error) {
  158. 158 for value.IsValid() && (value.Kind() == reflect.Interface || value.Kind() == reflect.Pointer) {
  159. 159 if value.IsNil() {
  160. 160 return nil, nil
  161. 161 }
  162. 162 value = value.Elem()
  163. 163 }
  164. 164 if !value.IsValid() {
  165. 165 return nil, nil
  166. 166 }
  167. 167
  168. 168 if shouldRoundTripJSONValue(value) {
  169. 169 return roundTripJSONValue(value.Interface())
  170. 170 }
  171. 171
  172. 172 switch value.Kind() {
  173. 173 case reflect.Struct:
  174. 174 return normalizeExpectedJSONObject(value)
  175. 175 case reflect.Map:
  176. 176 if value.IsNil() {
  177. 177 return nil, nil
  178. 178 }
  179. 179 return normalizeExpectedJSONMap(value)
  180. 180 case reflect.Slice, reflect.Array:
  181. 181 return normalizeExpectedJSONArray(value)
  182. 182 default:
  183. 183 return value.Interface(), nil
  184. 184 }
  185. 185}

  187. 187func normalizeExpectedJSONObject(value reflect.Value) (map[string]any, error) {
  188. 188 out := make(map[string]any)
  189. 189 typ := value.Type()
  190. 190 for i := 0; i < typ.NumField(); i++ {
  191. 191 field := typ.Field(i)
  192. 192 if field.PkgPath != "" {
  193. 193 continue
  194. 194 }
  195. 195
  196. 196 name, ok := jsonFieldName(field)
  197. 197 if !ok {
  198. 198 continue
  199. 199 }
  200. 200
  201. 201 fieldValue := value.Field(i)
  202. 202 if fieldValue.IsZero() {
  203. 203 continue
  204. 204 }
  205. 205
  206. 206 normalized, err := normalizeExpectedJSONValue(fieldValue)
  207. 207 if err != nil {
  208. 208 return nil, err
  209. 209 }
  210. 210 out[name] = normalized
  211. 211 }
  212. 212 return out, nil
  213. 213}

  215. 215func normalizeExpectedJSONMap(value reflect.Value) (map[string]any, error) {
  216. 216 out := make(map[string]any, value.Len())

  225. 225 normalized, err := normalizeExpectedJSONValue(value.MapIndex(key))
  226. 226 if err != nil {
  227. 227 return nil, err
  228. 228 }
  229. 229 out[name] = normalized
  230. 230 }
  231. 231 return out, nil
  232. 232}

  246. 246func shouldRoundTripJSONValue(value reflect.Value) bool {
  247. 247 jsonMarshalerType := reflect.TypeOf((*json.Marshaler)(nil)).Elem()
  248. 248 textMarshalerType := reflect.TypeOf((*encoding.TextMarshaler)(nil)).Elem()
  249. 249
  250. 250 typ := value.Type()
  251. 251 if typ.Implements(jsonMarshalerType) || typ.Implements(textMarshalerType) {
  252. 252 return true
  253. 253 }
  254. 254 return value.CanAddr() && (value.Addr().Type().Implements(jsonMarshalerType) || value.Addr().Type().Implements(textMarshalerType))
  255. 255}

  257. 257func roundTripJSONValue(value any) (any, error) {
  258. 258 body, err := json.Marshal(value)
  259. 259 if err != nil {
  260. 260 return nil, err
  261. 261 }
  262. 262 var decoded any
  263. 263 if err := json.Unmarshal(body, &decoded); err != nil {
  264. 264 return nil, err
  265. 265 }
  266. 266 return decoded, nil
  267. 267}

  269. 269func jsonFieldName(field reflect.StructField) (string, bool) {
  270. 270 tag := field.Tag.Get("json")
  271. 271 if tag == "-" {
  272. 272 return "", false
  273. 273 }
  274. 274 name, _, _ := strings.Cut(tag, ",")
  275. 275 if name == "" {
  276. 276 name = field.Name
  277. 277 }
  278. 278 return name, true
  279. 279}

  281. 281func jsonMapKeyString(value reflect.Value) (string, bool, error) {
  282. 282 for value.IsValid() && (value.Kind() == reflect.Interface || value.Kind() == reflect.Pointer) {
  283. 283 if value.IsNil() {
  284. 284 return "", false, nil
  285. 285 }
  286. 286 value = value.Elem()
  287. 287 }
  288. 288 if !value.IsValid() {
  289. 289 return "", false, nil
  290. 290 }
  291. 291
  292. 292 switch value.Kind() {
  293. 293 case reflect.String:
  294. 294 return value.String(), true, nil
  295. 295 case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
  296. 296 return strconv.FormatInt(value.Int(), 10), true, nil
  297. 297 case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64, reflect.Uintptr:
  298. 298 return strconv.FormatUint(value.Uint(), 10), true, nil
  299. 299 }
  300. 300
  301. 301 textMarshalerType := reflect.TypeOf((*encoding.TextMarshaler)(nil)).Elem()
  302. 302 if value.Type().Implements(textMarshalerType) || (value.CanAddr() && value.Addr().Type().Implements(textMarshalerType)) {
  303. 303 body, err := json.Marshal(value.Interface())
  304. 304 if err != nil {
  305. 305 return "", false, err
  306. 306 }
  307. 307 var decoded string
  308. 308 if err := json.Unmarshal(body, &decoded); err != nil {
  309. 309 return "", false, err
  310. 310 }
  311. 311 return decoded, true, nil
  312. 312 }
  313. 313
  314. 314 return "", false, fmt.Errorf("unsupported json map key type %s", value.Type())
