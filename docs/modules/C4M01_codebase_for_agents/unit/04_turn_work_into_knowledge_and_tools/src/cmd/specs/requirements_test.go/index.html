  10. 10func TestRequirementValidationPrecedesExecution(t *testing.T) {
  11. 11 for _, example := range []struct {
  12. 12 name string
  13. 13 contract string
  14. 14 links []spec.RequirementID
  15. 15 failure string
  16. 16 }{
  17. 17 {"unknown link", "W1. Known requirement.", []spec.RequirementID{"W2"}, `unknown requirement "W2"`},
  18. 18 {"missing link", "W1. Known requirement.", nil, "missing Covers requirements"},
  19. 19 {"duplicate link", "W1. Known requirement.", []spec.RequirementID{"W1", "W1"}, "duplicate requirement W1"},
  20. 20 {"duplicate requirement", "W1. First.\nW1. Second.", []spec.RequirementID{"W1"}, "line 2: duplicate requirement W1 (first at line 1)"},
  21. 21 {"malformed requirement", "W01. Invalid.", []spec.RequirementID{"W1"}, `invalid requirement ID "W01"`},
  22. 22 {"empty contract", "# No requirements", []spec.RequirementID{"W1"}, "no numbered requirements found"},
  23. 23 } {
  24. 24 t.Run(example.name, func(t *testing.T) {
  25. 25 executed := false
  26. 26 scenario := spec.New("invalid links", func(bld *spec.B) {
  27. 27 bld.When(spec.Get("/")).Then("invalid links prevent execution").Expect(spec.Custom(func(spec.Result) error {
  28. 28 executed = true
  29. 29 return nil
  30. 30 }))
  31. 31 })
  32. 32 scenario.Requirements = example.links
  33. 33 var out bytes.Buffer
  34. 34 code := runContract(t, nil, &out, []spec.Feature{{Name: "runner", Specs: []spec.Spec{scenario}}}, example.contract)
  35. 35 if code != 2 || executed || !strings.Contains(out.String(), example.failure) {
  36. 36 t.Fatalf("code=%d executed=%v, want validation failure %q:\n%s", code, executed, example.failure, out.String())
  37. 37 }
  38. 38 })
  39. 39 }

  42. 42func TestCoverageReportsAllInterpretationsInDocumentOrderAfterFailures(t *testing.T) {
  43. 43 features := []spec.Feature{{Name: "runner", Specs: []spec.Spec{
  44. 44 spec.New("passing", func(bld *spec.B) {
  45. 45 bld.When(spec.Get("/")).Then("the catalogue is readable").Covers("W1", "W2")
  46. 46 }),
  47. 47 spec.New("failing", func(bld *spec.B) {
  48. 48 bld.When(spec.Get("/missing")).Then("a missing page is readable").Covers("W2")
  49. 49 }),
  50. 50 spec.New("after failure", func(bld *spec.B) {
  51. 51 bld.When(spec.Get("/")).Then("execution continues").Covers("W1")
  52. 52 }),
  53. 53 }}}
  54. 54 var out bytes.Buffer
  55. 55 code := runContract(t, []string{"-coverage"}, &out, features, "# Behavior\n\nW2. Second.\n\nW1. First.\n\nW3. No linked interpretation.")
  56. 56 if code != 1 || !strings.Contains(out.String(), "3 scenarios: 2 passed, 1 failed") {
  57. 57 t.Fatalf("failed run lost its outcome: code=%d\n%s", code, out.String())
  58. 58 }
  59. 59 _, report, found := strings.Cut(out.String(), "REQUIREMENT")
  60. 60 if !found {
  61. 61 t.Fatalf("no coverage report:\n%s", out.String())
  62. 62 }
  63. 63 var rows []string
  64. 64 for _, line := range strings.Split(report, "\n")[1:] {
  65. 65 if line = strings.Join(strings.Fields(line), " "); line != "" {
  66. 66 rows = append(rows, line)
  67. 67 }
  68. 68 }
  69. 69 want := []string{
  70. 70 "W2 PASS runner / passing the catalogue is readable",
  71. 71 "W2 FAIL runner / failing a missing page is readable",
  72. 72 "W1 PASS runner / passing the catalogue is readable",
  73. 73 "W1 PASS runner / after failure execution continues",
  74. 74 "W3 NONE (no linked scenario) No linked interpretation.",
  75. 75 }
  76. 76 if strings.Join(rows, "\n") != strings.Join(want, "\n") {
  77. 77 t.Fatalf("coverage rows:\n%s\nwant:\n%s", strings.Join(rows, "\n"), strings.Join(want, "\n"))
  78. 78 }
